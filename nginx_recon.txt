# /etc/nginx/conf.d/recon.conf — Eko Recon: /recon static + /api proxy, on :80 (LAN) and :443 (public HTTPS, self-signed)
upstream recon_backend { server 127.0.0.1:8000; keepalive 16; }

server {
    listen 80;
    server_name _;
    absolute_redirect off;
    client_max_body_size 100m;

    location /api/ {
        proxy_pass http://recon_backend;
        proxy_http_version 1.1;
        proxy_set_header Connection         "";
        proxy_set_header Host               $host;
        proxy_set_header X-Real-IP          $remote_addr;
        proxy_set_header X-Forwarded-For    $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto  $scheme;
        proxy_read_timeout    300s;
        proxy_send_timeout    300s;
        proxy_request_buffering off;
    }

    location = /recon { return 301 /recon/; }

    location ^~ /recon/assets/ {
        root /var/www;
        try_files $uri =404;
        access_log off;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    location = /recon/index.html {
        root /var/www;
        add_header Cache-Control "no-cache";
    }

    location /recon/ {
        root /var/www;
        try_files $uri /recon/index.html;
    }
}
